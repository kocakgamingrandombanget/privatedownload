function sendEmailToAll(subject, plainBody, htmlBody) {
    var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet();
    var dataRange = sheet.getDataRange().getValues();
    var emails = {};
    for (var i = 1; i < dataRange.length; i++) {
        if (dataRange[i][4] === "APPROVED" && dataRange[i][2]) {
            emails[dataRange[i][2]] = true;
        }
    }
    
    var subs = getSubscribers();
    for (var j = 0; j < subs.length; j++) {
        emails[subs[j]] = true;
    }
    
    var uniqueEmails = Object.keys(emails);
    for (var k = 0; k < uniqueEmails.length; k++) {
        try { 
            var options = {};
            if (htmlBody) options.htmlBody = htmlBody;
            MailApp.sendEmail(uniqueEmails[k], subject, plainBody, options); 
        } catch(e) {}
    }
}

function generateAddonLicense(gamertag) {
    if (!gamertag) gamertag = "UNKNOWN";
    var salt = getSecretSalt(); 
    var str = String(gamertag).toLowerCase().trim() + salt;
    var hash1 = 5381; var hash2 = 52711;
    for (var i = 0; i < str.length; i++) { 
        hash1 = ((hash1 << 5) + hash1) + str.charCodeAt(i);
        hash2 = ((hash2 << 5) + hash2) ^ str.charCodeAt(i); 
    }
    hash1 = (hash1 >>> 0);
    hash2 = (hash2 >>> 0);
    var mix = ("00000000" + hash1.toString(16)).slice(-8) + ("00000000" + hash2.toString(16)).slice(-8);
    mix = mix.toUpperCase();
    var chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"; var result = "";
    for (var j = 0; j < 16; j++) { 
        result += chars.charAt((parseInt(mix.charAt(j), 16) + (j * 3) + str.length) % chars.length);
    }
    return result.substring(0,4) + "-" + result.substring(4,8) + "-" + result.substring(8,12) + "-" + result.substring(12,16);
}

    var formattedFixed = formatNotes(fixedNotes);
    var formattedRemoved = formatNotes(removedNotes);
    var formattedMaintenance = formatNotes(maintenanceNotes);

        payload.embeds[0].fields.push({ "name": "🐛 Bug Fixes", "value": "> *No bug fixes.*\n\u200B", "inline": false });

function doGet(e) {
  try {
    var action = e.parameter.action;
    if (action === "get_free_version") {
        return ContentService.createTextOutput(JSON.stringify({ "status": "success", "data": getFreeVersion() })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "get_portfolio") {
        return ContentService.createTextOutput(JSON.stringify({ "status": "success", "portfolio": getPortfolio() })).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === "get_servers") {
        return ContentService.createTextOutput(JSON.stringify({ "status": "success", "servers": getMemberServers() })).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === "get_stats") {
        var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet();
        var dataRange = sheet.getDataRange().getValues();
        var activeCount = 0;
        var membersList = [];
        for (var i = 1; i < dataRange.length; i++) { 
            if (dataRange[i][4] === "APPROVED") {
                activeCount++;
                var rawEmail = dataRange[i][2] ? String(dataRange[i][2]) : "";
                var maskedEmail = "-";
                if (rawEmail.indexOf("@") > -1) {
                    var parts = rawEmail.split("@");
                    maskedEmail = parts[0].substring(0, 3) + "***@" + parts[1];
                }
                
                membersList.push({
                    gamertag: dataRange[i][1],
                    email: maskedEmail
                });
            } 
        }
        membersList.reverse();
        return ContentService.createTextOutput(JSON.stringify({ 
            "status": "success", 
            "activeUsers": activeCount,
            "members": membersList
        })).setMimeType(ContentService.MimeType.JSON);
    }

    if (action === "get_changelog") return ContentService.createTextOutput(JSON.stringify({ "status": "success", "changelogs": getAddonHistory() })).setMimeType(ContentService.MimeType.JSON);
    
    var row = parseInt(e.parameter.row);
    if (action && row) {
      var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet();
      var gamertag = sheet.getRange(row, 2).getValue();
      var emailTarget = sheet.getRange(row, 3).getValue();
      var statusSaatIni = sheet.getRange(row, 5).getValue();
      var loginPassword = sheet.getRange(row, 6).getValue();
      if (statusSaatIni === "APPROVED" || statusSaatIni === "REJECTED") return createHtmlResponse("Cancelled", "Order has already been processed (" + statusSaatIni + ").", true);
      if (action === "approve") {
        sheet.getRange(row, 5).setValue("APPROVED");
        var addonLicenseCode = sheet.getRange(row, 8).getValue();
        if (!addonLicenseCode || addonLicenseCode === "") { 
            addonLicenseCode = generateAddonLicense(gamertag);
            sheet.getRange(row, 8).setValue(addonLicenseCode);
        }
        MailApp.sendEmail(emailTarget, "Heraclaus License Active!", "Hello " + gamertag + ",\n\nThank you for your purchase! Payment verified.\n\n--- LOGIN DETAILS ---\n🔗 Login: " + URL_CLIENT_PORTAL + "\n📧 Email: " + emailTarget + "\n🔑 Password: " + loginPassword + "\n\n--- IN-GAME LICENSE ---\n📜 Key: " + addonLicenseCode);
        return createHtmlResponse("Approved!", "Account activated and email sent.", false);
      } else if (action === "reject") {
        sheet.getRange(row, 5).setValue("REJECTED");
        return createHtmlResponse("Rejected", "Order rejected.", true);
      }
    }
    return createHtmlResponse("Error", "Invalid link.", true);
  } catch(error) { 
      return createHtmlResponse("Error", error.toString(), true); 
  }
